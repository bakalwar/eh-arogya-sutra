"""
E.H. Arogya Sutra Book PDF/Images se data nikalo.
Run: python extract_book.py
Input:  book/eh_arogya_sutra.pdf
book/pages/*.jpg (scanned pages)
Output: book/extracted/
├── medicines.json
├── anatomy.json
├── polarity.json
├── potency.json
├── electricity.json
└── symptoms.json
"""
import fitz          # pip install pymupdf
import pytesseract   # pip install pytesseract
import json, re, os
from PIL import Image
from pathlib import Path
BOOK_PDF    = "book/eh_arogya_sutra.pdf"
BOOK_PAGES  = "book/pages/"
OUTPUT_DIR  = "book/extracted/"
os.makedirs(OUTPUT_DIR, exist_ok=True)
# ══════════════════════════════════
# STEP 1: PDF se text nikalo
# ══════════════════════════════════
def extract_pdf_text() -> dict:
"""
PDF ke har page ka text nikalo.
Returns: {page_num: text}
"""
pages = {}
if not os.path.exists(BOOK_PDF):
print(f"PDF nahi mila: {BOOK_PDF}")
return {}
doc = fitz.open(BOOK_PDF)
print(f"Total pages: {len(doc)}")
for i, page in enumerate(doc):
# Direct text extraction (typed PDF)
text = page.get_text("text")
# Agar text kam ho to OCR karo
if len(text.strip()) < 50:
pix  = page.get_pixmap(dpi=300)
img  = Image.frombytes("RGB",
[pix.width, pix.height], pix.samples)
text = pytesseract.image_to_string(
img, lang="hin+eng",
config="--psm 6 --oem 3")
pages[i+1] = text
print(f"Page {i+1}: {len(text)} chars")
return pages
# ══════════════════════════════════
# STEP 2: Scanned Images se text
# ══════════════════════════════════
def extract_images_text() -> dict:
"""
Book ke scanned page images se text nikalo.
"""
pages = {}
img_dir = Path(BOOK_PAGES)
if not img_dir.exists():
return {}
images = sorted(img_dir.glob("*.jpg")) + \
sorted(img_dir.glob("*.png"))
for i, img_path in enumerate(images):
img  = Image.open(img_path)
# Hindi + English OCR
text = pytesseract.image_to_string(
img,
lang="hin+eng",
config="--psm 6 --oem 3 -c tessedit_char_whitelist=ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+-./: अआइईउऊएओकखगघचछजझटठडढणतथदधनपफबभमयरलवशषसह"
)
pages[i+1] = text
print(f"Image {img_path.name}: {len(text)} chars")
return pages

# STEP 3: Medicines Table nikalo
# ══════════════════════════════════
def extract_medicines(pages: dict) -> list:
"""
Book se har medicine ki information nikalo.
Pattern: S-1, C-4, A-2, P-1, F-1, VEN-1 etc.
"""
medicines = []
seen = set()
# Medicine patterns in EH book
MED_PATTERNS = [
r"S[-\s]?(\d+)",   # Scrofoloso
r"C[-\s]?(\d+)",   # Canceroso
r"A[-\s]?(\d+)",   # Angitico
r"P[-\s]?(\d+)",   # Pectorale
r"F[-\s]?(\d+)",   # Febrifugo
r"L[-\s]?(\d+)",   # Limfatico
r"VEN[-\s]?(\d+)", # Venereo
]
# Group descriptions from book
GROUP_MAP = {
"S": {"group":"Scrofoloso","system":"Lymphatic",
"hindi":"स्क्रोफोलोसो","nature":"Negative"},
"C": {"group":"Canceroso", "system":"Multi-system",
"hindi":"कैंसरोसो","nature":"Both"},
"A": {"group":"Angitico",  "system":"Sanguine/Blood",
"hindi":"एंजिटिको","nature":"Negative"},
"P": {"group":"Pectorale", "system":"Respiratory",
"hindi":"पेक्टोरले","nature":"Negative"},
"F": {"group":"Febrifugo", "system":"Nervous/Fever",
"hindi":"फेब्रीफुगो","nature":"Both"},
"L": {"group":"Limfatico", "system":"Blood/Immunity",
"hindi":"लिम्फेटिको","nature":"Both"},
"V": {"group":"Venereo",   "system":"Antiviral",
"hindi":"वेनेरेओ","nature":"Negative"},
}
for page_num, text in pages.items():
lines = text.split("\n")
for line in lines:
line_clean = line.strip()
if not line_clean: continue
# Check each medicine pattern
for prefix in ["S","C","A","P","F","L","VEN"]:
grp_key = prefix[0]
pat = rf"{prefix}[-\s]?(\d+)"
matches = re.findall(pat, line_clean, re.IGNORECASE)
for num in matches:
code = f"{prefix}-{num}"
if code in seen: continue
seen.add(code)
grp = GROUP_MAP.get(grp_key, {})
medicines.append({
"medicine_code": code,
"group_name":    grp.get("group", prefix),
"group_hindi":   grp.get("hindi", prefix),
"target_system": grp.get("system", ""),
"nature":        grp.get("nature", "Both"),
"page_found":    page_num,
"context_line":  line_clean[:200],
"action": extract_action(line_clean, code)
})
# Sort: S-1, S-2... C-1, C-2...
medicines.sort(key=lambda x: (
x["medicine_code"][0],
int(x["medicine_code"].split("-")[1])
if x["medicine_code"].split("-")[1].isdigit() else 0
))
print(f"Medicines found: {len(medicines)}")
return medicines
def extract_action(line: str, code: str) -> str:
"""Line se medicine ka kaam nikalo."""
# Remove medicine code from line
clean = re.sub(r"[A-Z]+-\d+", "", line).strip()
clean = re.sub(r"[-:;,]", " ", clean).strip()
if len(clean) > 20:
return clean[:200]
return "EH formula mein upyog"

# STEP 5: Polarity Rules nikalo
# ══════════════════════════════════
def extract_polarity_rules(pages: dict) -> list:
"""Book se Polarity Law data nikalo."""
rules = [
{"rule_id":"POL_001",
"condition":"Positive Rog",
"hindi_condition":"धनात्मक रोग",
"symptoms":["sujan","fever","bukhar","pain","swelling","high values"],
"medicine_type":"Negative Medicine",
"potency":["D6","D10","D30"],
"electricity":["Y.E.","G.E.","B.E."],
"rule":"Higher dilution = more negative effect",
"warning":"D4 KABHI NAHI in positive rog"},
{"rule_id":"POL_002",
"condition":"Negative Rog",
"hindi_condition":"ऋणात्मक रोग",
"symptoms":["kamzori","weakness","atrophy","low values","pale"],
"medicine_type":"Positive Medicine",
"potency":["D4","D6"],
"electricity":["R.E.","W.E."],
"rule":"Lower dilution = more positive effect",
"warning":"D30 KABHI NAHI in negative rog"},
]
# Book pages mein dhundho
POSITIVE_KEYWORDS = ["positive","dhanaatmak","badhna",
"swelling","inflammation","sujan"]
NEGATIVE_KEYWORDS = ["negative","rinatmak","ghatna",
"weakness","atrophy","kamzori"]
for page_num, text in pages.items():
text_low = text.lower()
pos_found = sum(1 for k in POSITIVE_KEYWORDS if k in text_low)
neg_found = sum(1 for k in NEGATIVE_KEYWORDS if k in text_low)
if pos_found + neg_found > 3:
print(f"Polarity rules found on page {page_num}")
# Extract relevant lines
for line in text.split("\n"):
if any(k in line.lower() for k in POSITIVE_KEYWORDS + NEGATIVE_KEYWORDS):
if len(line.strip()) > 20:
rules.append({
"rule_id": f"POL_BOOK_{page_num}",
"source_page": page_num,
"raw_text": line.strip()[:300],
"extracted": True
})
return rules

# STEP 8: Symptoms se Medicine Map
# ══════════════════════════════════
def extract_symptom_map(pages: dict) -> list:
"""
Book mein jahan bhi symptoms aur medicines
saath mein hain, woh extract karo.
"""
symptom_map = []
SYMPTOM_KEYWORDS = [
"fever","bukhar","headache","sir dard",
"joint","jod","kidney","gurda","liver",
"diabetes","sugar","asthma","blood pressure",
"weakness","kamzori","skin","twacha",
]
MED_PATTERN = r"[SCAPFLV][A-Z]*[-\s]\d+"
for page_num, text in pages.items():
text_low = text.lower()
lines = text.split("\n")
for line in lines:
line_low = line.lower().strip()
if not line_low: continue
# Check if symptom + medicine both present
has_symptom = any(s in line_low for s in SYMPTOM_KEYWORDS)
med_matches = re.findall(MED_PATTERN, line, re.IGNORECASE)
if has_symptom and len(med_matches) > 0:
# Find which symptom
matched_syms = [s for s in SYMPTOM_KEYWORDS if s in line_low]
symptom_map.append({
"symptoms": matched_syms[:3],
"medicines": med_matches[:5],
"page": page_num,
"raw_line": line.strip()[:300]
})
print(f"Symptom-Medicine mappings: {len(symptom_map)}")
return symptom_map

"""
Book se nikale gaye data ko summary mein use karo.
Jab doctor patient analyze kare, book ka data
relevant medicines ke baare mein aur jaankari de.
"""
import json, os
from database import SessionLocal
from models import EhMateriaMedica
OUTPUT_DIR = "book/extracted/"
def get_book_medicine_info(medicine_code: str) -> dict:
"""
Book se nikali gayi medicine information do.
Summary mein is data ko use karo.
"""
# First check extracted JSON
path = os.path.join(OUTPUT_DIR, "medicines.json")
if os.path.exists(path):
with open(path, encoding="utf-8") as f:
meds = json.load(f)
for m in meds:
if m.get("medicine_code") == medicine_code:
return {
"code": medicine_code,
"book_info": m.get("action",""),
"page": m.get("page_found",0),
"context": m.get("context_line",""),
"source": "book"
}
# Fallback to DB
db = SessionLocal()
med = db.query(EhMateriaMedica).filter(
EhMateriaMedica.medicine_code == medicine_code
).first()
db.close()
if med:
return {
"code": medicine_code,
"book_info": med.action or "",
"group": med.group_name,
"system": med.target_system,

}

def enrich_formula_with_book(formula: dict) -> dict:
"""
Formula ki har dawa mein book ka data add karo.
Yeh summary mein use hoga.
"""
enriched = formula.copy()
for key in ["formula_a","formula_b","formula_c","formula_d"]:
f = enriched.get(key, {})
if not f: continue
meds = f.get("medicines", [])
enriched_meds = []
for med in meds:
info = get_book_medicine_info(med)
enriched_meds.append({
"code": med,
"book_data": info.get("book_info",""),
"page": info.get("page",0)
})
f["medicines_detail"] = enriched_meds
enriched[key] = f
return enriched
def get_symptom_book_reference(symptom: str) -> list:
"""
Book mein is symptom ke liye kya likha hai?
"""
path = os.path.join(OUTPUT_DIR, "symptoms.json")
if not os.path.exists(path):
return []
with open(path, encoding="utf-8") as f:
data = json.load(f)
matches = []
sym_low = symptom.lower()
for item in data:
if any(s in sym_low or sym_low in s
for s in item.get("symptoms",[])):
matches.append({
"medicines": item.get("medicines",[]),
"page": item.get("page",0),
"book_text": item.get("raw_line","")[:200]
})
return matches[:3]
