// FIX 1: formatFormula() — RULE 2 (Spacing fix)
// 'S-1 + Y.E. D10' → 'S1 + YE D10' (clean)
// ════════════════════════════════════════════════════
function formatFormula(medicines, potency) {
  if (!medicines || medicines.length === 0) return '--';
  const arr = Array.isArray(medicines)
    ? medicines
    : String(medicines).split('+').map(x => x.trim());
  const clean = arr
    .map(m => String(m).replace(/-/g,'').replace(/\./g,'').trim())
    .filter(Boolean);
  const pot = String(potency || 'D10').split('/')[0].trim();
  return `${clean.join(' + ')} ${pot}`;
  // Output: 'S10 + A3 + YE D10' ✅ (spaced, no hyphens)
}
 
// ════════════════════════════════════════════════════
// FIX 2: getPotency() — RULE 3 (Correct Polarity)
// तीव्र/Acute = HYPER = D10 (Higher dilution)
// जीर्ण/Chronic = HYPO = D3/D4 (Lower dilution)
// ════════════════════════════════════════════════════
function getPotency(phase, polarity) {
  if (polarity === 'POSITIVE' || polarity === 'HYPER') {
    return ['ACUTE','SUB_ACUTE'].includes(phase) ? 'D10' : 'D30';
  }
  // NEGATIVE/HYPO → Lower dilution
  return ['ACUTE','SUB_ACUTE'].includes(phase) ? 'D4' : 'D3';
}

async function generateClinicalSummary(input) {
  const {
    age = 30, gender = 'Male', weight = 60,
    bp_systolic = 120, bp_diastolic = 80,
    symptoms = [], phase = 'ACUTE',
    polarity = 'POSITIVE',
    report_values = {},
    formulas = {}
  } = input;
 
  // FIX 2: Correct potency
  const potency = getPotency(phase, polarity);
 
  const rawSymptoms = symptoms.map(s =>
    typeof s === 'object' ? (s.name || s.hindi || '') : String(s)
  ).filter(Boolean);
  const cleanSymptoms = rawSymptoms.join(', ');
  const genderHi = gender === 'Male' ? 'पुरुष' : 'महिला';
  const isHyper = ['ACUTE','SUB_ACUTE'].includes(phase);
 
  // FIX 1: Format formulas
  const fa = formulas.formula_a || {};
  const fb = formulas.formula_b || {};
  const faFmt = formatFormula(fa.medicines, potency);
  const fbFmt = formatFormula(fb.medicines, potency);
 
  // Report values text
  const reportText = Object.entries(report_values)
    .filter(([k,v]) => v && v !== 0)
    .map(([k,v]) => `${k}: ${v}`)
    .join(', ') || 'उपलब्ध नहीं';
 
  // STEP 1: Database se 38 medicines fetch
  let databaseColumnsContent = '';
  try {
    const matchedMeds = await db.get38MedicinesBySymptoms(cleanSymptoms);
    databaseColumnsContent = matchedMeds
      .map(m => `औषधि: ${m.medicine_name} | क्लिनिकल कार्य: ${m.action_hindi}`)
      .join('\n');
  } catch (dbErr) {
    console.log('DB fetch error, using fallback mapping.');
    databaseColumnsContent = 'S1: लसीका शुद्धि | A1: रक्त संचार';
  }
 
  // STEP 2: Build structured prompt
  const userPrompt = `
रोगी क्लिनिकल केस:
- लक्षण: ${cleanSymptoms}
- आयु: ${age} वर्ष | लिंग: ${genderHi}
- रक्तचाप: ${bp_systolic}/${bp_diastolic} mmHg
- अवस्था: ${isHyper ? 'तीव्र (Acute/HYPER) — D10 आवश्यक' : 'जीर्ण (Chronic/HYPO) — D3/D4 आवश्यक'}
- प्रयोगशाला रिपोर्ट: ${reportText}
 
📋 डेटाबेस से प्राप्त 38 दवाओं का प्रमाणित विवरण:
${databaseColumnsContent}
 
AI द्वारा निर्धारित फॉर्मूला:
- फॉर्मूला A: ${faFmt}
- फॉर्मूला B: ${(fbFmt !== '--' ? fbFmt : 'उपलब्ध नहीं')}
- निर्धारित पोटेंसी: ${potency}
 
कृपया बिना किसी किताब का नाम या पेज नंबर लिखे,
शुद्ध देवनागरी हिंदी में 7-खंड विस्तृत प्रिस्क्रिप्शन
तैयार करें जैसा STRICT RULES में बताया गया है।
`;
 
  // STEP 3: Ollama call
  try {
    const response = await axios.post(
      'http://localhost:11434/api/generate',
      {
        model: 'llama3.2:3b', // FIX: updated model
        system: MATTEI_SYSTEM_PROMPT,
        prompt: userPrompt,
        stream: false,
        options: {
          temperature: 0.2,
          num_predict: 3000,
          top_p: 0.9
        }
      },
      { timeout: 120000 }
    );
 
    const aiText = response.data.response || '';
 
    // Check if response is substantial
    if (aiText.length > 400) {
      return {
        summary: aiText,
        source: 'ollama-38meds',
        potency: potency,
        ready: true
      };
    }
    throw new Error('Response too short');
 
  } catch (ollamaError) {
    console.log('Ollama unavailable, using rule-based fallback.');
 
    // FIX: Full rule-based fallback (not just 1 line!)
    return {
      summary: buildFallbackSummary(input, formulas, potency),
      source: 'rule-engine-fallback',
      potency: potency,
      ready: true
    };
  }
}
