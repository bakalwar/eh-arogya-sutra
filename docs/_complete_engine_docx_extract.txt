// UTIL 1: Formula Spacing Fix
// 'YED10' → 'YE D10' | 'S10A3' chips → clean
// ═══════════════════════════════════════════════════
function normalizeFormulaSpacing(str) {
  if (!str) return '--';
  return str
    .replace(/(S\d+|A\d+|C\d+|F\d+|L\d+|P\d+|V\d+)(D\d+)/g, '$1 $2')
    .replace(/(RE|BE|YE|WE|GE)(D\d+)/g, '$1 $2')
    .replace(/\s+/g, ' ').trim();
}

// UTIL 2: Formula Builder
// ['S-1','A-3','Y.E.'] + 'D10' → 'S1 + A3 + YE D10'
// ═══════════════════════════════════════════════════
function buildFormula(medicines, potency) {
  if (!medicines || medicines.length === 0) return '--';
  const arr = Array.isArray(medicines) ? medicines
    : String(medicines).split('+').map(x => x.trim());
  const clean = arr
    .map(m => String(m).replace(/-/g,'').replace(/\./g,'').trim())
    .filter(Boolean);
  const pot = String(potency || 'D10').split('/')[0].trim();
  return `${clean.join(' + ')} ${pot}`;
}

// STEP 1: POLARITY + POTENCY ENGINE
// Score-based detection — more accurate than if/else
// ═══════════════════════════════════════════════════
function determinePolarity(input) {
  const {
    bp_systolic=120, bp_diastolic=80,
    symptoms=[], phase='ACUTE', report_values={}
  } = input;
 
  const symText = symptoms
    .map(s => typeof s==='object' ? (s.name||'') : String(s))
    .join(' ').toLowerCase();
 
  let posScore = 0, negScore = 0;
 
  // ── BP Score ──
  if (bp_systolic >= 180) posScore += 8;  // Crisis
  else if (bp_systolic >= 140) posScore += 5;
  else if (bp_systolic < 90)  negScore += 6;
  else if (bp_systolic < 100) negScore += 4;
 
  // ── Symptom Score ──
  const HYPER_W = ['sujan','fever','bukhar','pain','dard','swelling',
    'infection','uric','jalan','redness','hot','high','tez'];
  const HYPO_W  = ['kamjori','weakness','kamzori','chakkar','sunn',
    'numbness','thakan','atrophy','low','pale','cold','anemia','bejaan'];
 
  HYPER_W.forEach(w => { if (symText.includes(w)) posScore += 2; });
  HYPO_W.forEach(w  => { if (symText.includes(w)) negScore += 2; });
 
  // ── Report Values Score ──
  const RANGES = {
    hemoglobin:[12,17], wbc:[4000,11000], sugar:[70,100],
    creatinine:[0.6,1.2], uric_acid:[3.5,7], potassium:[3.5,5],
    pth:[15,65], hba1c:[4,5.6], sgpt:[7,56]
  };
  Object.entries(report_values).forEach(([k,v]) => {
    const r = RANGES[k];
    if (!r || !v) return;
    if (v > r[1]) posScore += 3;  // HIGH = Hyper
    if (v < r[0]) negScore += 3;  // LOW  = Hypo
  });
 
  // ── Phase Score ──
  if (['ACUTE','SUB_ACUTE'].includes(phase)) posScore += 2;
  if (['CHRONIC','DEGENERATIVE'].includes(phase)) negScore += 2;
 
  const isHyper = posScore >= negScore;
 
  // ── Potency by Phase + Polarity ──
  let potency = 'D10';
  if (isHyper) {
    if (phase === 'ACUTE')        potency = 'D10';
    else if (phase === 'SUB_ACUTE') potency = 'D10';
    else if (phase === 'CHRONIC') potency = 'D30';
    else potency = 'D30';
  } else {
    if (phase === 'CHRONIC')      potency = 'D3';
    else potency = 'D4';
  }
 
  return {
    isHyper,
    polarity: isHyper ? 'POSITIVE' : 'NEGATIVE',
    polarityState: isHyper
      ? 'HYPER (अति-सक्रिय / उग्र अवस्था)'
      : 'HYPO (निष्क्रिय / शिथिल अवस्था)',
    potency, posScore, negScore
  };
}

function determineElectricity(symText, isHyper, bp_systolic) {
  // Default
  let electricity = isHyper
    ? 'BE (नीली विद्युत)' : 'RE (लाल विद्युत)';
  let location = 'प्रभावित अंगों पर सूती कपड़े की पट्टी (Compress) 20 मिनट।';
 
  // SCIATICA
  if (symText.includes('sciatica') || symText.includes('saitica')) {
    electricity = isHyper ? 'GE (हरी विद्युत)' : 'RE (लाल विद्युत)';
    location = 'रीढ़ की हड्डी के निचले हिस्से (L4-S1) से प्रभावित पैर के टखने तक, जहाँ-जहाँ दर्द हो।';
  }
  // HEADACHE / VERTIGO / MIGRAINE
  else if (['chakkar','headache','migraine','vertigo','sir'].some(w => symText.includes(w))) {
    electricity = 'WE (सफेद विद्युत)';
    location = 'सिर के पिछले भाग (Occiput) और दोनों कनपटी पर हल्के हाथ से।';
  }
  // JOINT PAIN / URIC / GOUT
  else if (['joint','jod','uric','gout','arthritis','ghutna'].some(w => symText.includes(w))) {
    electricity = isHyper ? 'GE (हरी विद्युत)' : 'RE (लाल विद्युत)';
    location = 'प्रभावित जोड़ों पर — घुटना, कूल्हा, टखना — 20 मिनट Compress।';
  }
  // HEART / HIGH BP / CARDIAC
  else if (bp_systolic >= 140 || ['heart','dhadkan','cardiac','bp'].some(w => symText.includes(w))) {
    electricity = 'BE (नीली विद्युत)';
    location = 'माथे पर और हृदय के पास (बाईं छाती पर) ठंडे पानी में भीगा कपड़ा 20 मिनट।';
  }
  // SKIN / KHUJLI / RASH
  else if (['skin','twacha','khujli','rash','daad','eczema'].some(w => symText.includes(w))) {
    electricity = isHyper ? 'BE (नीली विद्युत)' : 'RE (लाल विद्युत)';
    location = 'प्रभावित त्वचा क्षेत्र पर सीधे Compress या हल्की मालिश।';
  }
  // LUNG / SAANS / COUGH
  else if (['lung','saans','khansi','cough','chest','asthma'].some(w => symText.includes(w))) {
    electricity = 'YE (पीली विद्युत)';
    location = 'छाती के अग्र भाग और पीठ पर — फेफड़ों के दोनों ओर Compress।';
  }
  // KIDNEY / URINARY
  else if (['kidney','gurda','peshaab','urine','pathri'].some(w => symText.includes(w))) {
    electricity = isHyper ? 'GE (हरी विद्युत)' : 'RE (लाल विद्युत)';
    location = 'पीठ पर गुर्दे के स्थान (कमर के दोनों ओर) पर 20 मिनट Compress।';
  }
  // WEAKNESS / GENERAL HYPO
  else if (!isHyper) {
    electricity = 'RE (लाल विद्युत)';
    location = 'प्रभावित अंग पर RE Compress — ऊर्जा प्रवाह बहाल करने के लिए।';
  }
 
  return { electricity, location };
}

async function generateClinicalSummary(input) {
  const {
    age=30, gender='Male', weight=60,
    bp_systolic=120, bp_diastolic=80,
    symptoms=[], phase='ACUTE',
    report_text='', report_values={},
    formulas={}
  } = input;
 
  // Sab symptom text ek jagah
  const rawSym = symptoms.map(s =>
    typeof s==='object' ? (s.name||'') : String(s));
  if (report_text) rawSym.push(report_text);
  const fullText  = rawSym.join(' ').toLowerCase();
  const genderHi  = gender==='Male' ? 'पुरुष' : 'महिला';
 
  // STEP 1: Polarity + Potency
  const polData = determinePolarity(input);
  const { potency, isHyper, polarityState } = polData;
 
  // STEP 2: Electricity + Location
  const elecData = determineElectricity(fullText, isHyper, bp_systolic);
  const { electricity, location } = elecData;
 
  // STEP 3: Formula strings
  const fa    = formulas?.formula_a || {};
  const fb    = formulas?.formula_b || {};
  const faFmt = buildFormula(fa.medicines, potency);
  const fbFmt = buildFormula(fb.medicines, potency);
 
  // STEP 4: Report values text
  const reportText = Object.entries(report_values)
    .filter(([k,v])=>v&&v!==0)
    .map(([k,v])=>`${k}: ${v}`)
    .join(', ') || 'उपलब्ध नहीं';
 
  // STEP 5: Database se 38 medicines
  let bookData = '';
  try {
    const meds   = await db.get38MedicinesBySymptoms(fullText);
    bookData = meds
      .map(m => `- **${m.medicine_name}:** ${m.action_hindi}`)
      .join('\n');
  } catch (e) {
    console.log('DB error, using defaults:', e.message);
    bookData = '- S10: पाचन तंत्र शुद्धि\n- A1: रक्त संचार नियंत्रण';
  }
 
  // STEP 6: Build Ollama prompt
  const userPrompt = `
रोगी क्लिनिकल केस:
- लक्षण एवं रिपोर्ट: ${fullText}
- आयु: ${age} वर्ष | लिंग: ${genderHi} | भार: ${weight} किग्रा
- रक्तचाप: ${bp_systolic}/${bp_diastolic} mmHg
- अवस्था: ${phase==='ACUTE'?'तीव्र (HYPER — D10)':'जीर्ण (HYPO — D3/D4)'}
- ध्रुवता: ${polarityState}
- पोटेंसी: ${potency}
- प्रयोगशाला रिपोर्ट: ${reportText}
 
AI फॉर्मूला:
- मिश्रण A: ${faFmt}
- मिश्रण B: ${(fbFmt!=='--'?fbFmt:'उपलब्ध नहीं')}
 
बाह्य विद्युत:
- विद्युत: ${electricity}
- स्थान: ${location}
 
📋 38 दवाओं का प्रमाणित विवरण:
${bookData}
 
कृपया बिना किताब का नाम या पेज नंबर लिखे,
शुद्ध देवनागरी हिंदी में 7-खंड विस्तृत प्रिस्क्रिप्शन
तैयार करें। 1000+ शब्द। कोई खंड अधूरा नहीं।
`;
 
  // STEP 7: Ollama Call
  try {
    const resp = await axios.post(
      'http://localhost:11434/api/generate',
      {
        model: 'llama3.2:3b',      // Updated
        system: MATTEI_SYSTEM_PROMPT, // Separate system
        prompt: userPrompt,
        stream: false,
        options: { temperature:0.2, num_predict:3500, top_p:0.9 }
      },
      { timeout: 120000 }
    );
 
    const aiText = resp.data.response || '';
    const clean  = normalizeFormulaSpacing(aiText);
 
    if (clean.length > 500) {
      return {
        summary: clean,
        source:  'ollama-38meds',
        potency, electricity,
        ready: true
      };
    }
    throw new Error('Response too short');
 
  } catch (err) {
    console.log('Ollama unavailable → rule-based fallback');
 
    // FULL 7-section fallback
    return {
      summary: buildFallbackSummary(
        input, polData, elecData, formulas, potency),
      source:  'rule-engine-fallback',
      potency, electricity,
      ready: true
    };
  }
}
