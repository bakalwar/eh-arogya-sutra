function formatFormula(medicines, potency) {
  if (!medicines || medicines.length === 0)
    return '--';
 
  // Remove hyphens: S-1 → S1, C-4 → C4
  const cleanMeds = medicines.map(m => {
    if (!m) return '';
    return String(m)
      .replace(/-/g, '')    // S-1 → S1
      .replace(/\./g, '')  // Y.E. → YE
      .trim();
  }).filter(Boolean);
 
  // Join with ' + '
  const medStr = cleanMeds.join(' + ');
 
  // Potency sirf ek baar aakhir mein
  const pot = (potency || 'D10')
    .replace(/D(\d+)\/D(\d+)/, 'D$1');
 
  return `${medStr} ${pot}`;
}

  phase, polarity) {
 
  // RULE 3: Correct mapping
  if (polarity === 'POSITIVE') {
    // Hyper/Positive rog → Negative dose
    // = Higher dilution
    if (phase === 'ACUTE')        return 'D10';
    if (phase === 'SUB_ACUTE')    return 'D10';
    if (phase === 'CHRONIC')      return 'D30';
    if (phase === 'DEGENERATIVE') return 'D30';
    return 'D10';
  }
 
  if (polarity === 'NEGATIVE') {
    // Hypo/Negative rog → Positive dose
    // = Lower dilution
    if (phase === 'ACUTE')     return 'D4';
    if (phase === 'SUB_ACUTE') return 'D4';
    if (phase === 'CHRONIC')   return 'D3';
    return 'D4';
  }
 
  return 'D10'; // Default
}
